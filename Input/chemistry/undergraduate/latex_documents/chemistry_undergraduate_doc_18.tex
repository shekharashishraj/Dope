\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Considering 0.1 M aqueous solutions of each of the following, which solution has the lowest pH?
\begin{enumerate}[label=(\alph*)]
    \item Na$_{2}$CO$_{3}$
    \item Na$_{3}$PO$_{4}$
    \item Na$_{2}$S
    \item NaCl
\end{enumerate}
\item Calculate the Larmor frequency for a proton in a magnetic field of 1 T.
\begin{enumerate}[label=(\alph*)]
    \item 23.56 GHz
    \item 42.58 MHz
    \item 74.34 kHz
    \item 13.93 MHz
\end{enumerate}
\item Of the following solutions, which will have the highest ionic strength? (Assume complete dissociation.)
\begin{enumerate}[label=(\alph*)]
    \item 0.050 M AlCl 3
    \item 0.100 M NaCl
    \item 0.050 M CaCl 2
    \item 0.100 M HCl
\end{enumerate}
\item Of the following ionic substances, which has the greatest lattice enthalpy?
\begin{enumerate}[label=(\alph*)]
    \item MgO
    \item MgS
    \item NaF
    \item NaCl
\end{enumerate}
\item The 13 C spectrum of which isomer of C$_{6}$H$_{14}$ has lines with three distinct chemical shifts?
\begin{enumerate}[label=(\alph*)]
    \item hexane
    \item 2-methylpentane
    \item 3-methylpentane
    \item 2,3-dimethylbutane
\end{enumerate}
\end{enumerate}
