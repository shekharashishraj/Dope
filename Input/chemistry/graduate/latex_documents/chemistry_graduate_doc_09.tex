\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Calculate the Larmor frequency for a proton in a magnetic field of 1 T.
\begin{enumerate}[label=(\alph*)]
    \item 23.56 GHz
    \item 50 MHz
    \item 100 MHz
    \item 42.58 MHz
    \item 1 GHz
\end{enumerate}
\item Calculate the normality of a solution 40 ml of which is required to neutralize 0.56 g of KOH.
\begin{enumerate}[label=(\alph*)]
    \item 0.5 N
    \item 0.2 N
    \item 0.75 N
    \item 0.1 N
    \item 1 N
\end{enumerate}
\item Each resonance form of the nitrate ion, NO 3-, has how many sigma and how many pi bonds?
\begin{enumerate}[label=(\alph*)]
    \item 4 sigma and 1 pi
    \item 3 sigma and 1 pi
    \item 0 sigma and 4 pi
    \item 3 sigma and 2 pi
    \item 2 sigma and 1 pi
\end{enumerate}
\item In a chemical reaction requiring two atoms of phosphorus for five atoms of oxygen, how many grams of oxygen are required by 3.10 g of phosphorus?
\begin{enumerate}[label=(\alph*)]
    \item 9.30 g O
    \item 2.50 g O
    \item 4.00 g O
    \item 5.00 g O
    \item 3.10 g O
\end{enumerate}
\item A sample of $255 \mathrm{mg}$ of neon occupies $3.00 \mathrm{dm}^3$ at $122 \mathrm{K}$. Use the perfect gas law to calculate the pressure of the gas.
\begin{enumerate}[label=(\alph*)]
    \item $4.20 \times 10^{-2}$ $\text{atm}$
    \item $5.55 \times 10^{-2}$ $\text{atm}$
    \item $6.33 \times 10^{-2}$ $\text{atm}$
    \item $3.15 \times 10^{-2}$ $\text{atm}$
    \item $8.21 \times 10^{-2}$ $\text{atm}$
\end{enumerate}
\end{enumerate}
