\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item What pressure is required to compress 5 liters of gas at 1 atm. pressure to 1 liter at a constant temperature?

    \item 9 atm
    \item 5 atm
    \item 6 atm
    \item 10 atm

\item Calculate the percentage difference between $e^x$ and $1+x$ for $x=0.0050$
\begin{enumerate}[label=(\alph*)]
    \item 7.50 $10^{-3} \%$
    \item 2.50 $10^{-3} \%$
    \item 0.75 $10^{-3} \%$
    \item 2.00 $10^{-3} \%$
    \item 1.25 $10^{-3} \%$
\end{enumerate}
\item A stock solution of 12.0 M sulfuric acid is made available. What is the best procedure to make up 100. mL of 4.0 M sulfuric acid using the stock solution and water prior to mixing?
\begin{enumerate}[label=(\alph*)]
    \item Add 67.7 mL of 12.0 M acid to the flask, and then dilute it with 33.3 mL of water.
    \item Add 33.3 mL of water to the flask, and then add 66.7 mL of 12.0 M acid.
    \item Add 40 mL of water to the flask, and then add 60 mL of 12.0 M acid.
    \item Add 60 mL of 12.0 M acid to the flask, and then dilute it with 40 mL of water.
    \item Add 50 mL of 12.0 M acid to the flask, and then dilute it with 50 mL of water.
\end{enumerate}
\item Assuming complete ionization, calculate (a) the pH of 0.0001 NHCl, .(b) thepOHof 0.0001 N KOH.

    \item pH is 10, pOH is 4
    \item pH is 4, pOH is 10
    \item pH is 3, pOH is 3
    \item pH is 4, pOH is 4
    \item pH is 1, pOH is 1
\end{enumerate}
\item Calculate the energy of a hydrogen atom in its ground state (n = 1) using the mass of the electron rather than the reduced mass of the hydrogen atom.

    \item 1.6022 \ensuremath{\times} 10^-19 C
    \item - 2.1802 \ensuremath{\times} 10^-18 J
    \item 9.1095 \ensuremath{\times} 10^-31 kg
\end{enumerate}
\end{enumerate}
