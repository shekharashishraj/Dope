\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Solve the equation 14 = w + 23.
\begin{enumerate}[label=(\alph*)]
    \item 9
    \item 37
    \item -23
    \item 14
    \item -14
\end{enumerate}
\item When the expression $-2 x^2-20 x-53$ is written in the form $a(x+d)^2+e$, where $a$, $d$, and $e$ are constants, what is the sum $a+d+e$?
\begin{enumerate}[label=(\alph*)]
    \item 0
    \item 2
    \item 6
    \item 80
    \item 20
\end{enumerate}
\item Use mental math to solve 8 y = 56.
\begin{enumerate}[label=(\alph*)]
    \item 8
    \item 10
    \item 4
    \item 14
    \item 6
\end{enumerate}
\item Suppose X and Y are random variables with E(X) = 312, var(X) = 6, E(X) = 307, and var(Y) = 8. What are the expected value and variance of the random variable X + Y?
\begin{enumerate}[label=(\alph*)]
    \item E(X + Y) = 307, var(X + Y) = 10
    \item E(X + Y) = 307, var(X + Y) = 6
    \item E(X + Y) = 619, var(X + Y) = 14
    \item E(X + Y) = 619, var(X + Y) = 7
    \item E(X + Y) = 312, var(X + Y) = 14
\end{enumerate}
\item Suppose today is Wednesday. What day of the week will it be $10^(10$^(10)) days from now?
\begin{enumerate}[label=(\alph*)]
    \item Sunday
    \item Wednesday
    \item Tuesday
    \item Saturday
    \item The day of the week cannot be determined
\end{enumerate}
\end{enumerate}
