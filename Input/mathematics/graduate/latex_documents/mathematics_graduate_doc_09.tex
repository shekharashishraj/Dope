\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item There are 72 students on the dance team, and 4 over 9 take tap lessons. How many students take tap lessons?
\begin{enumerate}[label=(\alph*)]
    \item 32
    \item 45
    \item 40
    \item 49
    \item 63
\end{enumerate}
\item A tournament has 393 players with 3 players on each team. Which expression shows how to find the number of teams in the tournament?
\begin{enumerate}[label=(\alph*)]
    \item 3 + 393
    \item 3 \ensuremath{\times} 393
    \item (393 - 3) \ensuremath{\times} 3
    \item 393 \ensuremath{\div} 3
    \item (393 + 3) \ensuremath{\div} 3
\end{enumerate}
\item Coloring the edges of a complete graph with n vertices in 2 colors (red and blue), what is the smallest n that guarantees there is either a 4-clique in red or a 4-clique in blue?
\begin{enumerate}[label=(\alph*)]
    \item 12
    \item 15
    \item 14
    \item 16
    \item 22
\end{enumerate}
\item The quadratic $2 x^2+5 x+12=19-7 x$ has two solutions. What is the positive difference between these solutions?
\begin{enumerate}[label=(\alph*)]
    \item 2\ensuremath{\sqrt{3}}
    \item \ensuremath{\sqrt{7}}
    \item 3\ensuremath{\sqrt{2}}
    \item 5\ensuremath{\sqrt{2}}
    \item 2
\end{enumerate}
\item What is the measure, in degrees, of an angle that is equivalent to 1/360 of a circle?
\begin{enumerate}[label=(\alph*)]
    \item 0
    \item 720
    \item 30
    \item 180
    \item 90
\end{enumerate}
\end{enumerate}
