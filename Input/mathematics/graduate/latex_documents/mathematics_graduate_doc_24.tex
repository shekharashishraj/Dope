\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item How many integers between 1 (included) and 100 (included) are divisible by either 2, 3, or 5?
\begin{enumerate}[label=(\alph*)]
    \item 82
    \item 60
    \item 80
    \item 68
    \item 74
\end{enumerate}
\item For how many positive integers k does the ordinary decimal representation of the integer k! end in exactly 99 zeros?
\begin{enumerate}[label=(\alph*)]
    \item Twenty
    \item Three
    \item None
    \item Four
    \item Five
\end{enumerate}
\item For a\ensuremath{\geq} 0, we define $S_a={x | dist(x, S) \leq a}$, where $dist(x,S)=inf_{y\in S}||x-y||$. Suppose S is convex. Is S\_a convex? Return 1 for yes and 0 for no.
\begin{enumerate}[label=(\alph*)]
    \item Undefined
    \item 0.0
    \item None of the above
    \item 1.0
\end{enumerate}
\item At West Elementary School, there are 20 more girls than boys. If there are 180 girls, how can you find the number of boys?
\begin{enumerate}[label=(\alph*)]
    \item divide 20 by 180
    \item add 180 to 20
    \item multiply 180 by 20
    \item multiply 180 by 2
    \item divide 180 by 20
\end{enumerate}
\item We know that $y'=(x+y) / 2$, we also know that $y(x=0) = 2, y(x=0.5) = 2.636, y(x=1) = 3.595, y(x=1.5) = 4.9868$, what is the value of y(2) using Adams bashforth predictor method.

    \item 5.4321
    \item 7.1234
    \item 6.2456
    \item 7.7654
    \item 5.6789
\end{enumerate}
\end{enumerate}
