\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Which of the following sets has the greatest cardinality?
\begin{enumerate}[label=(\alph*)]
    \item The set of all finite subsets of R
    \item R
    \item The set of all infinite subsets of R
    \item The set of all functions from Z to Z
    \item The set of all functions from \{0, 1\} to R
\end{enumerate}
\item There are 25 suitcases, 5 of which are damaged. Three suitcases are selected at random. What is the probability that exactly 2 are damaged?
\begin{enumerate}[label=(\alph*)]
    \item 2/50
    \item 1/25
    \item 1/30
    \item 3/75
    \item 4/100
\end{enumerate}

    \item \$0.50
    \item \$0.20
    \item \$0.60
    \item \$0.10
\end{enumerate}
\item What number makes the equation 35 / ? = 7 true?

    \item 25
    \item 7
    \item 6
    \item 10
