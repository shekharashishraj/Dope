\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Let \{N(t), t \ensuremath{\in} [0, \ensuremath{\infty})\} be a Poisson process with rate of $\lambda = 4$ and $X_1$ be the first arrival time. Given N(t) = 1, then what is $P(X_1 <= t / 2)$?
\begin{enumerate}[label=(\alph*)]
    \item 0.5
    \item 1
    \item 0.95
    \item 0.65
    \item 0.85
\end{enumerate}
\item 15. Consider the initial value problem $ 4 y^{\prime \prime}+12 y^{\prime}+9 y=0, \quad y(0)=1, \quad y^{\prime}(0)=-4 . $\$ Determine where the solution has the value zero.
\begin{enumerate}[label=(\alph*)]
    \item 0.8
    \item 0.3
    \item 0.1
    \item 0.5
    \item 0.4
\end{enumerate}
\item Statement 1 | A factor group of a non-Abelian group is non-Abelian. Statement 2 | If K is a normal subgroup of H and H is a normal subgroup of G, then K is a normal subgroup of G.
\begin{enumerate}[label=(\alph*)]
    \item False, False
    \item True, True
    \item Not Determinable, True
    \item Not Determinable, False
    \item True, False
\end{enumerate}
\item Sam paid \$8.28 for 18 stamps. At this rate, how much would it cost Sam to buy 12 stamps?
\begin{enumerate}[label=(\alph*)]
    \item \$4.52
    \item \$7.28
    \item \$5.52
    \item \$6.28
    \item \$3.72
\end{enumerate}
\item The graph of the equation $x + 2 y + 3 = 0$ is perpendicular to the graph of the equation $ax + 2 y + 3 = 0$. What is the value of $a$?

    \item -1
    \item -13
    \item -4
    \item -3
    \item 0
\end{enumerate}
\end{enumerate}
