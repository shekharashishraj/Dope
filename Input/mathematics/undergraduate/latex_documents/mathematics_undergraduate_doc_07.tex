\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item If the finite group G contains a subgroup of order five but no element of G other than the identity is its own inverse, then the order of G could be
\begin{enumerate}[label=(\alph*)]
    \item 8
    \item 20
    \item 30
    \item 35
\end{enumerate}
\item If A is a subset of the real line R and A contains each rational number, which of the following must be true?
\begin{enumerate}[label=(\alph*)]
    \item If A is open, then A = R.
    \item If A is closed, then A = R.
    \item If A is uncountable, then A = R.
    \item If A is uncountable, then A is open.
\end{enumerate}
\item What is the volume of the solid in xyz-space bounded by the surfaces y = $x^2$, y = 2 - $x^2$, z = 0, and z = y + 3?
\begin{enumerate}[label=(\alph*)]
    \item 8/3
    \item 16/3
    \item 32/3
    \item 104/105
\end{enumerate}
\item Let x\_1 = 1 and x\_(n+1) = sqrt(3+2 x\_n) for all positive integers n. If it is assumed that \{x\_n\} converges, then lim x\_n =
\begin{enumerate}[label=(\alph*)]
    \item 3
    \item e
    \item sqrt(5)
    \item 0
\end{enumerate}
\item Statement 1 | A ring homomorphism is one to one if and only if the kernel is \{0\}. Statement 2 | Q is an ideal in R.
\begin{enumerate}[label=(\alph*)]
    \item True, True
    \item False, False
    \item True, False
    \item False, True
\end{enumerate}
\end{enumerate}

\section*{True / False (10 marks)}
\textit{Answer True or False}
\begin{enumerate}[label=\arabic*.]
\setcounter{enumi}{5}
\item True or False: Every ring must contain a multiplicative identity, which is not true for all rings.
\item True or False: Every group of order 42 has a normal subgroup of order 7 due to the Sylow theorems.
\item True or False: In the ring of continuous real-valued functions on [0, 1], the product of two nonzero functions can be zero.
\item True or False: If f is a linear transformation and f(1, 1) = 1 and f(-1, 0) = 2, then f(3, 5) equals 9.
\item True or False: The unity of a subring must be the same as the unity of the larger ring it belongs to.
\end{enumerate}
