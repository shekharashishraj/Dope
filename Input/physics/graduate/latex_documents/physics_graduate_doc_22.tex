\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item A chain of mass m per unit length and total length l lies in a pile on the floor. At time t = 0 a force P is applied and the chain is raised with constant velocity v. Find P(t).
\begin{enumerate}[label=(\alph*)]
    \item P = mv(t + g)
    \item P = m(v + $gt)^2$
    \item P =mgvt+ $mv^2$ =mv(v +gt)
    \item P = $mv^2$ - mgvt
    \item P = mv + gt
\end{enumerate}
\item How much heat is produced in 5 minutes by an electric iron which draws 5 amperes from a 120-volt line?

    \item 80,000 calories
    \item 36,000 calories
    \item 65,000 calories
    \item 48,000 calories
\end{enumerate}
