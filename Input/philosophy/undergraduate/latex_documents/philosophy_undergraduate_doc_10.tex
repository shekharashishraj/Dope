\begin{enumerate}[label=(\alph*)]
    \item lie whenever doing so is necessary to prevent a catastrophe.
    \item always do whatever brings about the greatest happiness.
    \item always assert what you don't think to be the case.
    \item all of the above.
\end{enumerate}

\begin{enumerate}[label=(\alph*)]

    \item Nature.
    \item the universe.
\end{enumerate}

\begin{enumerate}[label=(\alph*)]

    \item people who knowingly buy stolen goods.
    \item thieves who steal from the poor.
    \item merchants who sell harmful products.
    \item war profiteers exploiting conflict.
\end{enumerate}

    \item pursue knowledge.
    \item pursue happiness.
    \item focus on material wealth.
    \item none of the above.
\end{enumerate}
\end{enumerate}
