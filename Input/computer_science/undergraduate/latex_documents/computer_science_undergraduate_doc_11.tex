\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Which types of functions grow the slowest?
\begin{enumerate}[label=(\alph*)]
    \item O(log $e^N$)
    \item O(log N)
    \item O(log log N)
    \item O(N)

\item Church's thesis equates the concept of "computable function" with those functions computable by, for example, Turing machines. Which of the following is true of Church's thesis?
\begin{enumerate}[label=(\alph*)]
    \item It was first proven by Alan Turing.
    \item It has not yet been proven, but finding a proof is a subject of active research.
    \item It can never be proven.
    \item It is now in doubt because of the advent of parallel computers.
\end{enumerate}
\item Let f(X) = if x = 1 then 0 else [x * f(x - 1) + x**2]. The value of f(4) is
\begin{enumerate}[label=(\alph*)]
    \item 53
    \item 29
    \item 50
    \item 100
\end{enumerate}
\item What is the output of "abc"[::-1] in Python 3?
\begin{enumerate}[label=(\alph*)]
    \item Error
    \item abc
    \item cba
    \item c
\end{enumerate}
\item The boolean expression a[i] == max || !(max != a[i]) can be simplified to
\begin{enumerate}[label=(\alph*)]
    \item a[i] == max
    \item a[i] != max
    \item a[i] \textless{} max || a[i] \textgreater{} max
    \item FALSE
\end{enumerate}
\end{enumerate}
