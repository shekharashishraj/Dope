\textbf{Multiple Choice Questions (Single correct)}\\[0.5em]
\emph{Instructions: Select the correct option with a short explanation.}

\begin{enumerate}[label=Q\arabic*., leftmargin=*]
\item In a standard RNN, what is the primary purpose of the ``hidden state''?\\[0.25em]
\textbf{A.} To store the final output of the entire sequence.\\
\textbf{B.} To act as a memory that carries information from previous time steps\\
\textbf{C.} To calculate the loss function for the current word\\
\textbf{D.} To normalize the input embeddings

\item Why is the Softmax function typically used in the final layer of a Neural Language Model?\\[0.25em]
\textbf{A.} To ensure the output values sum to 1, representing a probability distribution\\
\textbf{B.} To prevent the gradients from exploding\\
\textbf{C.} To increase the speed of the training process\\
\textbf{D.} To allow the model to handle variable-length sequences

\item What is a key limitation of feed-forward neural networks for NLP tasks?\\[0.25em]
\textbf{A.} They cannot use embeddings\\
\textbf{B.} They assume fixed input size and struggle with temporal order\\
\textbf{C.} They cannot be trained with gradient descent\\
\textbf{D.} They do not use activation functions

\item What is the purpose of the ``Forward Pass'' in training?\\[0.25em]
\textbf{A.} To update weights based on the loss\\
\textbf{B.} To calculate the predicted output and compute the loss\\
\textbf{C.} To unroll the network into a sequence\\
\textbf{D.} To calculate the gradient for each parameter

\item What is true about the weights $U$, $V$, and $W$ across different timestamps in an RNN?\\[0.25em]
\textbf{A.} They are randomly re-initialized at every step\\
\textbf{B.} They are reused across timestamps\\
\textbf{C.} They increase in value as the sequence progresses\\
\textbf{D.} Only the output weight $V$ is reused

\item Suppose all activation functions in a 3-layer neural network are linear. What is the effective model equivalent to?\\[0.25em]
\textbf{A.} A deeper network with more representational power\\
\textbf{B.} A single linear model\\
\textbf{C.} A recurrent neural network\\
\textbf{D.} A decision tree

\item When training an RNN for language modeling using teacher forcing, the model predicts:\\[0.25em]
\textbf{A.} The next word given the predicted history\\
\textbf{B.} The next word given the ground-truth history\\
\textbf{C.} The current word given itself\\
\textbf{D.} All words simultaneously

\item In an RNN, the hidden state at time $t$ depends on:\\[0.25em]
\textbf{A.} Only the input at time $t$\\
\textbf{B.} Only the hidden state at time $t-1$\\
\textbf{C.} Both the input at time $t$ and the hidden state at time $t-1$\\
\textbf{D.} All future inputs

\item Consider a neural network layer defined as: $h = f(Wx)$. The input vector $x$ has shape $(n_0 \times 1)$, and the output vector $h$ has shape $(n_1 \times 1)$. What must be the shape of $W$?\\[0.25em]
\textbf{A.} $(n_0 \times n_1)$\\
\textbf{B.} $(n_1 \times n_0)$\\
\textbf{C.} $(n_0 \times 1)$\\
\textbf{D.} $(1 \times n_1)$

\item What is the main role of backpropagation in neural network training?\\[0.25em]
\textbf{A.} To compute predictions during inference\\
\textbf{B.} To calculate gradients of the loss with respect to parameters\\
\textbf{C.} To initialize weights randomly\\
\textbf{D.} To normalize input data

\end{enumerate}
