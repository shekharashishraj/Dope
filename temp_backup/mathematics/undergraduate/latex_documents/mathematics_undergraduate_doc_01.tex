\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item What is the units digit in the standard decimal expansion of the number 7\^{}25?
\begin{enumerate}[label=(\alph*)]
    \item 1
    \item 3
    \item 5
    \item 7
\end{enumerate}
\item Find all zeros in the indicated finite field of the given polynomial with coefficients in
that field. x\^{}3 + 2x + 2 in Z\_7
\begin{enumerate}[label=(\alph*)]
    \item 1
    \item 2
    \item 2,3
    \item 6
\end{enumerate}
\item If f(z) is an analytic function that maps the entire finite complex plane into the real
axis, then the imaginary axis must be mapped onto
\begin{enumerate}[label=(\alph*)]
    \item the entire real axis
    \item a point
    \item a ray
    \item an open finite interval
\end{enumerate}
\item In the group G = \{2, 4, 6, 8) under multiplication modulo 10, the identity element is
\begin{enumerate}[label=(\alph*)]
    \item 6
    \item 8
    \item 4
    \item 2
\end{enumerate}
\item Let (Z\_\{10\} , +, x) be the ring of integers modulo 10, and let S be the subset of Z\_\{10\}
represented by \{0, 2, 4, 6, 8\}. Which of the following statements is FALSE?
\begin{enumerate}[label=(\alph*)]
    \item (S, +, x) is closed under addition modulo 10.
    \item (S, +, x) is closed under multiplication modulo 10.
    \item (S, +, x) has an identity under addition modulo 10.
    \item (S, +, x) has no identity under multiplication modulo 10.
\end{enumerate}
\end{enumerate}
