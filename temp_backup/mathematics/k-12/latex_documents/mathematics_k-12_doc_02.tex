\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item A grocerystore sells a bag of 5 lemons for \$2.00. What is the unit cost of each lemon in
the bag?
\begin{enumerate}[label=(\alph*)]
    \item \$2.50
    \item \$0.60
    \item \$0.40
    \item \$0.10
\end{enumerate}
\item The mean of 7 numbers is 15. When an 8th number is added, the mean decreases to 12. What
is the 8th number?
\begin{enumerate}[label=(\alph*)]
    \item -12
    \item -9
    \item 0
    \item 8
\end{enumerate}
\item In parallelogram $ABCD$, angle $B$ measures $110^\circ$. What is the number of degrees in
the measure of angle $C$?
\begin{enumerate}[label=(\alph*)]
    \item 145
    \item 90
    \item 60
    \item 70
\end{enumerate}
\item If all the values of a data set are the same, all of the following must equal zero except
for which one?
\begin{enumerate}[label=(\alph*)]
    \item Mean
    \item Standard deviation
    \item Variance
    \item Range
\end{enumerate}
\item What expression can be used to show 270,240 written in expanded form?
\begin{enumerate}[label=(\alph*)]
    \item 200,000 + 7,000 + 200 + 4
    \item 200,000 + 7,000 + 200 + 40
    \item 200,000 + 70,000 + 200 + 40
    \item 200,000 + 70,000 + 2,000 + 40
\end{enumerate}
\end{enumerate}
