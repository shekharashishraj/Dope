\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Which ratio CANNOT form a proportion with 8 over 18?
\begin{enumerate}[label=(\alph*)]
    \item 46 over 108
    \item 12 over 27
    \item 60 over 135
    \item 4 over 9
\end{enumerate}
\item Three points are chosen randomly and independently on a circle. What is the probability
that all three pairwise distances between the points are less than the radius of the
circle?
\begin{enumerate}[label=(\alph*)]
    \item \ensuremath{\frac{1}{20}}
    \item \ensuremath{\frac{1}{32}}
    \item \ensuremath{\frac{1}{12}}
    \item \ensuremath{\frac{1}{16}}
\end{enumerate}
\item We roll a fair 6-sided die 5 times. What is the probability that we get a 6 in at most 2
of the rolls?
\begin{enumerate}[label=(\alph*)]
    \item \ensuremath{\frac{125}{648}}
    \item \ensuremath{\frac{25}{648}}
    \item \ensuremath{\frac{625}{648}}
    \item \ensuremath{\frac{1}{648}}
\end{enumerate}
\item The first number in a number pattern is 28. The pattern rule is to add 14 to get the next
number in the pattern. If the pattern continues, which statement is true?
\begin{enumerate}[label=(\alph*)]
    \item All the numbers in the pattern can be divided equally by 10.
    \item All the numbers in the pattern can be divided equally by 4.
    \item All the numbers in the pattern can be divided equally by 8.
    \item All the numbers in the pattern can be divided equally by 7.
\end{enumerate}
\item What is the sum of all integer solutions to $|n| < |n-3| < 9$?
\begin{enumerate}[label=(\alph*)]
    \item -9
    \item -14
    \item 14
    \item 9
\end{enumerate}
\end{enumerate}
