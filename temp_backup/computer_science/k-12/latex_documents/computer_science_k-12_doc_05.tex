\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item If a, b, and c are integers, which of the following conditions is sufficient to guarantee
that the expression a \textless{} c || a \textless{} b \&\& !(a == c) evaluates to true?
\begin{enumerate}[label=(\alph*)]
    \item a \textless{} c
    \item a \textless{} b
    \item a \textgreater{} b
    \item a == b
\end{enumerate}
\item Which is a valid expression in Python 3.5?
\begin{enumerate}[label=(\alph*)]
    \item sort('ab')
    \item sorted('ab')
    \item "ab".sort()
    \item 1/0
\end{enumerate}
\item Which types of functions grow the slowest?
\begin{enumerate}[label=(\alph*)]
    \item O(N\^{}(1/2))
    \item O(N\^{}(1/4))
    \item O(N\^{}(1/N))
    \item O(N)
\end{enumerate}
\item A large Java program was tested extensively, and no errors were found. What can be
concluded?
\begin{enumerate}[label=(\alph*)]
    \item All of the preconditions in the program are correct.
    \item All of the postconditions in the program are correct.
    \item The program may have bugs.
    \item Every method in the program may safely be used in other programs.
\end{enumerate}
\item Which is the largest asymptotically?
\begin{enumerate}[label=(\alph*)]
    \item O(1)
    \item O(n)
    \item O(n\^{}2)
    \item O(log n)
\end{enumerate}
\end{enumerate}
