\section*{Multiple Choice (22 marks)}
\begin{enumerate}[label=\arabic*.]
\item Which of the following sentence is FALSE regarding regression?
\begin{enumerate}[label=(\alph*)]
    \item It may be used for interpretation.
    \item It is used for prediction.
    \item It relates inputs to outputs.
    \item It discovers causal relationships
\end{enumerate}
\item Consider the Bayesian network given below. How many independent parameters would we need if we made no assumptions about independence or conditional...
\begin{enumerate}[label=(\alph*)]
    \item 8
    \item 15
    \item 10
    \item 3
    \item 12
\end{enumerate}
\item If N is the number of instances in the training dataset, nearest neighbors has a classification run time of
\begin{enumerate}[label=(\alph*)]
    \item O( N!)
    \item O( N log N )
    \item O( N\^{}2 )
    \item O( N )
    \item O(1)
\end{enumerate}
\item Which of the following pairs of 8-bit, two's-complement numbers will result in overflow when the members of the pairs are added?
\begin{enumerate}[label=(\alph*)]
    \item 11110000, 00001111
    \item 01111111, 10000000
    \item 00000001, 10000000
    \item 10000001, 10101010
    \item 11111111, 00000001
\end{enumerate}
\item Which of the following has the greatest potential for compromising a user's personal privacy?
\begin{enumerate}[label=(\alph*)]
    \item A group of cookies stored by the user's Web browser
    \item The Wi-Fi network the user is connected to
    \item The user's search engine history
    \item The Internet Protocol (IP) address of the user's computer
    \item The operating system of the user's computer
\end{enumerate}
\item Which of the following is true of mutation-based fuzzing?
\begin{enumerate}[label=(\alph*)]
    \item It only works for network-based fuzzing, not file-based fuzzing
    \item It modifies the target program's code to test its robustness
    \item It requires an initial error-free program to start the process
    \item It uses a random number generator to create new inputs
    \item It uses a fixed set of inputs to test the program
\end{enumerate}
\item Calculate the Hamming pairwise distances and determine the minimum Hamming distance among the following codewords: 00000,10101,01010
\begin{enumerate}[label=(\alph*)]
    \item 3
    \item 6
    \item 10
    \item 7
    \item 8
\end{enumerate}
\item What's the maximum number of edges in a simple triangle free planar graph with 30 vertices?
\begin{enumerate}[label=(\alph*)]
    \item 69
    \item 60
    \item 63
    \item 84
    \item 45
\end{enumerate}
\item How many trees are there on n (n \textgreater{} 1) labeled vertices with no vertices of degree 1 or 2?
\begin{enumerate}[label=(\alph*)]
    \item 2n
    \item 2\^{}(n-1)
    \item 1
    \item 2
    \item n + 1
\end{enumerate}
\item Of the following, which gives the best upper bound for the value of f(N) where f is a solution to the recurrence f(2N + 1) = f(2N) = f(N) + log N for N \textgreater{}= 1,...
\begin{enumerate}[label=(\alph*)]
    \item O(N log(log N))
    \item O(N\^{}2)
    \item O(N)
    \item O(log N)
    \item O((log N)\^{}2)
\end{enumerate}
\item When does a buffer overflow occur, generally speaking?
\begin{enumerate}[label=(\alph*)]
    \item when copying a buffer from the stack to the heap
    \item when writing to a pointer that has been freed
    \item when the program runs out of memory
    \item when a pointer is used to access memory not allocated to it
    \item when a buffer is not initialized before being used
\end{enumerate}
\end{enumerate}
