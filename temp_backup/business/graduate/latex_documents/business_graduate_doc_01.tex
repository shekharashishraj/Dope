\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Mr. Carson signed a note as payment for a debt on March 15. If the note had a term of two
months, what was its maturity date?
\begin{enumerate}[label=(\alph*)]
    \item August 15
    \item May 15
    \item April 15
    \item March 15
    \item May 14
\end{enumerate}
\item Round .1476 to (a) 2 decimal places (b) 3 decimal places (c) 1 decimal place.
\begin{enumerate}[label=(\alph*)]
    \item (a) .14, (b) .147, (c) .1
    \item (a) .16, (b) .149, (c) .1
    \item (a) .15, (b) .148, (c) .1
    \item (a) .15, (b) .147, (c) .2
\end{enumerate}
\item A loan is made on December 17 for a time period of 120 days. When is it due (not a leap
year)?
\begin{enumerate}[label=(\alph*)]
    \item April 15
    \item April 13
    \item April 18
    \item May 15
    \item April 16
\end{enumerate}
\item A sofa has a markup of 20\% based on its selling price. Find the percent markup based on
the cost.
\begin{enumerate}[label=(\alph*)]
    \item 28\%
    \item 15\%
    \item 22\%
    \item 10\%
    \item 30\%
\end{enumerate}
\item Find the interest charge on a \$600 loan at 6\% for 60 days. Find the total amount due.
\begin{enumerate}[label=(\alph*)]
    \item \$612
    \item \$600
    \item \$620
    \item \$624
    \item \$606
\end{enumerate}
\end{enumerate}
