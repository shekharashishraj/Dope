\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Find the median in the set of data 23, 13, 18, 29, 32, 25.
\begin{enumerate}[label=(\alph*)]
    \item 18
    \item 24
    \item 25
    \item 29
\end{enumerate}
\item What is three fifth of 100?
\begin{enumerate}[label=(\alph*)]
    \item 3
    \item 5
    \item 20
    \item 60
\end{enumerate}
\item A circle passes through the points (3, 4) and (5, 7). Which of the following points CANNOT
lie on the circle?
\begin{enumerate}[label=(\alph*)]
    \item (-2, -1)
    \item (-1, -2)
    \item (5, 5)
    \item (6, 4)
\end{enumerate}
\item What is the largest factor of 54,321 that is less than 54,321?
\begin{enumerate}[label=(\alph*)]
    \item 6
    \item 121
    \item 18107
    \item 54,320
\end{enumerate}
\item How many ways are there to choose 3 cards from a standard deck of 52 cards, if all three
cards must be of different suits? (Assume that the order of the cards does not matter.)
\begin{enumerate}[label=(\alph*)]
    \item 8788
    \item 2
    \item 2197
    \item 21
\end{enumerate}
\end{enumerate}
