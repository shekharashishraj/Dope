\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Consider $x(t)$ to be given as, $ x(t)=10 \cos (20 \pi-\pi / 4)-5 \cos (50 \pi t) $$ What
is minimum sampling rate (/Hz) such that $y(t)=x(t)\$ ?
\begin{enumerate}[label=(\alph*)]
    \item 30
    \item 20
    \item 40
    \item 50
    \item 35
\end{enumerate}
\item Which types of functions grow the slowest?
\begin{enumerate}[label=(\alph*)]
    \item O(N)
    \item O(log log N)
    \item O(N!)
    \item O(log N)
    \item O(N\^{}N)
\end{enumerate}
\item A 3-way, set-associative cache is
\begin{enumerate}[label=(\alph*)]
    \item only effective when more than 3 processes are running alternately on the processor
    \item possible only with write-through
    \item one in which each main memory word can be stored at any of 4 cache locations
    \item possible only with write-back
    \item effective only if 3 or fewer processes are running alternately on the processor
\end{enumerate}
\item Let x = 8. What is x\textgreater{}\textgreater{}1 in Python 3?
\begin{enumerate}[label=(\alph*)]
    \item 5
    \item 3
    \item 0
    \item 8
    \item 4
\end{enumerate}
\item In Python 3, b = [11,13,15,17,19,21]; print(b[::2]) outputs what?
\begin{enumerate}[label=(\alph*)]
    \item [13,17,21]
    \item [13,15,17]
    \item [11,15]
    \item [11,13,15,19]
    \item [11,15,19]
\end{enumerate}
\end{enumerate}
