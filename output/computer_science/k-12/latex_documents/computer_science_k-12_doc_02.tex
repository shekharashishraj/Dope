\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item In Python 3, which of the following function sets the integer starting value used in
generating random numbers?
\begin{enumerate}[label=(\alph*)]
    \item choice(seq)
    \item randrange ([start,] stop [,step])
    \item random()
    \item seed([x])
\end{enumerate}
\item In Python 3, what is the following function returns item from the list with max value?
\begin{enumerate}[label=(\alph*)]
    \item cmp(list)
    \item len(list)
    \item max(list)
    \item min(list)
\end{enumerate}
\item Which is the smallest asymptotically?
\begin{enumerate}[label=(\alph*)]
    \item O(1)
    \item O(n)
    \item O(n\^{}2)
    \item O(log n)
\end{enumerate}
\item In Python 3, which of the following operator in python performs exponential (power)
calculation on operands?
\begin{enumerate}[label=(\alph*)]
    \item **
    \item //
    \item is
    \item not in
\end{enumerate}
\item Which types of functions grow the slowest?
\begin{enumerate}[label=(\alph*)]
    \item O(log e\^{}N)
    \item O(log N)
    \item O(log log N)
    \item O(N)
\end{enumerate}
\end{enumerate}
