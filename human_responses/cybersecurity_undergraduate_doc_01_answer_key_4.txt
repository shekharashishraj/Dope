16. Sample answer:

def count_divisible(numbers, m, n):
    return len(list(filter(lambda x: x % m == 0 or x % n == 0, numbers)))

# Example usage:
numbers = [10, 20, 33, 46, 55]
m = 5
n = 11
count = count_divisible(numbers, m, n)
print(count)  # Output will vary based on the list and numbers chosen
