16. Sample answer:

def find_numbers_greater_than_n(numbers, n):
 return list(filter(lambda x: x > n, numbers))

# Example
numbers = [1, 5, 8, 12, 20]
n = 10
result = find_numbers_greater_than_n(numbers, n)
print(result) # Output: [12, 20]
