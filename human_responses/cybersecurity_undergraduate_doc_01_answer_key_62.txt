16. Sample answer:

def find_divisible(numbers, m, n):
  return list(filter(lambda x: x % m == 0 or x % n == 0, numbers))

# Example usage:
numbers = [10, 20, 30, 40, 50]
m = 5
n = 10
divisible_numbers = find_divisible(numbers, m, n)
print(divisible_numbers)
