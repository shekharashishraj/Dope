11. Sample answer:

def is_palindrome(number):
    original = str(number)
    reversed_number = original[::-1]
    return original == reversed_number

# Example usage:
print(is_palindrome(121))  # True
print(is_palindrome(123))  # False

from datetime import datetime

def sort_tuple_by_date(tuples):
    return sorted(tuples, key=lambda x: datetime.strptime(x[1], '%Y-%m-%d'))

# Example usage:
example_tuples = [(1, '2023-01-01'), (2, '2022-12-31'), (3, '2023-01-02')]
sorted_tuples = sort_tuple_by_date(example_tuples)
print(sorted_tuples)  # [(2, '2022-12-31'), (1, '2023-01-01'), (3, '2023-01-02')]
