16. Sample answer:

def find_greater_than(numbers, n):
    return list(filter(lambda x: x > n, numbers))

# Example usage:
numbers_list = [1, 5, 8, 10, 12]
n = 7
print(find_greater_than(numbers_list, n))
# Output: [8, 10, 12]
